% Geometry for non-table presentation entries. These reproduce the horizontal
% text positions of the standard two-column CV table: default longtable padding
% places the body column at label width + 3\tabcolsep, with a 2\tabcolsep gap
% between the right-aligned label and the body text.
\newlength{\cvpresbodyoffset}
\setlength{\cvpresbodyoffset}{\dimexpr\cvlabelwidth+3\tabcolsep\relax}

\newlength{\cvpreslabelgap}
\setlength{\cvpreslabelgap}{\dimexpr2\tabcolsep\relax}

\newlength{\cvpresitemsep}
\setlength{\cvpresitemsep}{0.25em}

% Generic two-column CV section.
% Usage: \begin{CVList}{Section Title}[heading-space][arraystretch] ... \end{CVList}
\NewDocumentEnvironment{CVList}{m O{5px} O{1.2}}{%
	\renewcommand*{\arraystretch}{#3}%
	\begin{longtable}{U{\cvlabelwidth} T{\cvbodywidth}}
		& \multicolumn{1}{l}{\footnotesize{#1}} \\[#2]
	}{%
	\end{longtable}
}

% Labeled item: left-column label plus right-column content.
% Optional argument controls extra vertical space after the row.
\NewDocumentCommand{\CVItem}{O{} m m}{%
	#2 & #3 \ifstrempty{#1}{\\}{\\[#1]}%
}

% Unlabeled item: preserves the same left indent without printing a label.
\NewDocumentCommand{\CVBlankItem}{O{} m}{%
	& #2 \ifstrempty{#1}{\\}{\\[#1]}%
}

% Presentation sections are set as aligned hanging entries rather than table
% rows. The P-number is placed in the same label column used elsewhere, and the
% title/venues start at the same body-column x-position as the other CV lists.
% Since the label is printed with \llap at the start of the title line, it
% always shares the title baseline.
\NewDocumentEnvironment{CVPresentationList}{m O{0px}}{%
	\par\vspace{14pt}\noindent\hspace*{\cvpresbodyoffset}\footnotesize{#1}\par\vspace{#2}%
}{%
	\par
}

\NewDocumentCommand{\CVPresentation}{m m +m}{%
	\par\begingroup
	\normalsize
	\leftskip=\cvpresbodyoffset
	\parindent=0pt
	\noindent\llap{\makebox[\cvlabelwidth][r]{#1}\hspace{\cvpreslabelgap}}``#2''\par
	#3%
	\par\endgroup\vspace{\cvpresitemsep}%
}

% Venue item inside a presentation. The optional argument is for symbols such as
% $^\dagger$; empty city/year arguments are suppressed.
\NewDocumentCommand{\CVVenue}{O{} m m m}{%
	\noindent#1\textit{#2}\ifstrempty{#3}{}{ $\cdot$ #3}\ifstrempty{#4}{}{ #4}\par%
}

% Four-column teaching table.

	}{%
	\end{longtable}
}
